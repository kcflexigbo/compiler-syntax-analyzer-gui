int main()
{
    double denom[100],a;
    int k=0;
    int num[100];

    cout<<"Enter the denominations below(Enter -1 to exit): "<<endl;
    for(int i=0;i<100; i++)
    {
        cout<<"\ndenom "<<i+1<<":";
        cin>>a;
        if(a==-1) break;
        if(a<0)
        {
            cout<<"\nError Please Retry"<<endl;
            i--;
            continue;
        }
        if(searcher(denom,a,k)==1)
        {
            cout<<"\nError Denomination duplication detected"<<endl;
            i--;
            continue;
        }
        denom[i]=a;
        k++;
    }

    sorter(denom,k);
    initializer(num);

    cout<<"\nThe denominations entered are: "<<endl;
    display(denom,k);

    double amount;
    inputter(amount);
    cout<<"\nEnterred Amount is: "<<amount<<endl;

    bool status= coinChanger(amount,denom,num,k);
    display(status,num,denom,k,amount);
}

void initializer(int num[])
{
    for(int i=0; i<100; i++)
    {
        num[i]=0;
    }
}

void sorter(double arr[], int k)
{
    int a;
    for(int i=0; i<k-1; i++)
    {
        a=i;
        for(int j=i+1; j<k; j++)
        {
            if(arr[a]>arr[j]) a=j;
        }
        if(a!=i) swapper(i,a,arr);
    }
}

void swapper(int a, int b, double arr[])
{
    arr[a]= arr[a] + arr[b];
    arr[b]= arr[a] - arr[b];
    arr[a]= arr[a] - arr[b];
}

void display(double arr[], int k)
{
    for(int i=0; i<k; i++)
    {
        if(i==k-1) {cout<<arr[i]<<"."<<endl;break;}
        cout<<arr[i]<<",";
    }
}

{
    for(int i=0; i<k; i++)
    {
        if(i==k-1) {cout<<arr[i]<<"."<<endl;break;}
        cout<<arr[i]<<",";

int searcher(double arr[], double n, int k)
{
    for(int i=0; i<k; i++)
    {
        if(arr[i]==n) return 1;
    }
    return 0;
}

int inputter(double &amount)
{
    double a;
    cout<<"\nPlease Input the amount: ";
    cin>>a;
    if(a<=0) 
    {
        cout<<"\nError. Please Retry"<<endl;
        inputter(amount);
        return 0;
    }
    amount=a;
    return 1;
}

bool coinChanger(double amount,double denom[], int num[], int k)
{
    int t=1;
    int p=k-1;
    double namount=amount;

    while(namount>0)
    {
        if(t!=1) return false;
        if(namount>=denom[p])
        {
            namount-=denom[p];
            num[p]+=1;
        }else if(p!=0)
        {
            p--;
        }else if (namount<=0.001){
            return true;
        }else{
            t=0;
        }
    }
    return true;
}

void display(bool status, int n[],double arr[], int k, double amount)
{
    if(status==false)
    {
        cout<<"\nThe Coin Cannot be changed using the Values given"<<endl;
        display(n,arr,k);
    }else{
        cout<<"\nThe amount "<<amount<<" has been changed using the denominations to ";
        display(n,arr,k);
        cout<<".\n";
    }
}

void display(int n[], double arr[], int k)
{
    for(int i=k; i>=0; i--)
    {
        if(n[i]==0) continue;
        cout<<arr[i]<<"("<<n[i]<<") , ";
    }
}
